import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from seaborn import load_dataset

# -----------------------------------
# Load Titanic Dataset
# -----------------------------------

dataset = pd.read_csv("train.csv")

print("First 5 Rows of Titanic Dataset:")
print(dataset.head())

# -----------------------------------
# Box Plot: Gender vs Age
# -----------------------------------

sns.boxplot(
    x='Sex',
    y='Age',
    data=dataset
)

plt.title(
    "Box Plot of Age by Gender"
)

plt.show()

# -----------------------------------
# Box Plot with Survival Hue
# -----------------------------------

sns.boxplot(
    x='Sex',
    y='Age',
    data=dataset,
    hue='Survived'
)

plt.title(
    "Age Distribution by Gender and Survival"
)

plt.show()
